Before (current)

      if (recvNo.toLowerCase() !== 'skip') {
        await updateSheetLink(recvNo, category, fileId, billNo, branch, dealer, amount);
      }

After (new logic)

      if (recvNo.toLowerCase() === 'skip' && category === 'recv-pos') {
        // recv-pos always has a RECV number — ask again
        await redisSet(`recv_pending:${userId}`, recvPending, 300);
        await sendLineReply(replyToken, '⚠️ recv-pos ต้องมีเลข RECV เสมอ กรุณาพิมพ์เลข RECV:');
        return;
      }
      if (recvNo.toLowerCase() !== 'skip') {
        await updateSheetLink(recvNo, category, fileId, billNo, branch, dealer, amount);
      } else {
        // invoice / shipping-cost skip — append row with รอเลข placeholder
        await updateSheetLink('รอเลข', category, fileId, billNo, branch, dealer, amount);
      }
